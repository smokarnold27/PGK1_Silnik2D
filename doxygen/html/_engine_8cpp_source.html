121void Engine::Run()
122{
123 PrimitiveRenderer renderer;
124
125 // Inicjalizacja bufora i jednorazowe narysowanie wszystkich figur statycznych na starcie
126 renderer.ClearBuffer(WHITE);
127 for (const auto& obj : objects) {
128 if (!dynamic_cast<Player*>(obj.get())) {
129 obj->Draw(renderer);
130 }
131 }
132 // Rysowanie wszystkich bitmap w buforze
133 for (const auto& bmp : bitmap_objects) {
134 if (!dynamic_cast<Player*>(bmp.get())) {
135 bmp->Draw(renderer);
136 }
137 }
138
139 // Bufor pomocniczy przechowujący wycinek tła pod graczem
140 const int player_size = 64;
141 std::vector<Color> underPlayerBuffer(player_size * player_size, WHITE);
142
143 while (is_running && !WindowShouldClose())
144 {
145 float delta_time = GetFrameTime();
146 if (IsKeyPressed(KEY_ESCAPE)) {
147 is_running = false;
148 }
149
150 // Obsługa wypełnień (w tym momencie w buforze nie ma gracza, więc wypełniamy czyste tło/figury)
151 if (IsMouseButtonPressed(MOUSE_BUTTON_LEFT)) {
152 Vector2 mouse = GetMousePosition();
153 renderer.FloodFill(static_cast<int>(mouse.x), static_cast<int>(mouse.y), YELLOW);
154 }
155 if (IsMouseButtonPressed(MOUSE_BUTTON_RIGHT)) {
156 Vector2 mouse = GetMousePosition();
157 renderer.BorderFill(static_cast<int>(mouse.x), static_cast<int>(mouse.y), SKYBLUE, BLACK);
158 }
159
160 // Aktualizacja pozycji gracza
161 int px = 0, py = 0;
162 if (player_ptr) {
163 player_ptr->Update(delta_time);
164 Point2D pos = player_ptr->GetPosition();
165 px = pos.GetX();
166 py = pos.GetY();
167
168 // Zapisanie fragmentu tła, na którym zaraz narysujemy gracza
169 for (int dy = 0; dy < player_size; ++dy) {
170 for (int dx = 0; dx < player_size; ++dx) {
171 underPlayerBuffer[dy * player_size + dx] = renderer.GetPixelColor(px + dx, py + dy);
172 }
173 }
174
175 // Narysowanie gracza do bufora pikseli PRZED wywołaniem RenderBuffer()
176 player_ptr->Draw(renderer);
177 }
178
179 // Wyświetlenie bufora (tło + gracz) na ekranie
180 BeginDrawing();
181 ClearBackground(WHITE);
182 renderer.RenderBuffer();
183 EndDrawing();
184
185 // Przywrócenie oryginalnego tła pod graczem (usuwa ślad gracza przed kolejną klatką)
186 if (player_ptr) {
187 for (int dy = 0; dy < player_size; ++dy) {
188 for (int dx = 0; dx < player_size; ++dx) {
189 renderer.PutPixel(px + dx, py + dy, underPlayerBuffer[dy * player_size + dx]);
190 }
191 }
192 }
193 }
194}
